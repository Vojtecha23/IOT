

from tkinter import Tk, Scale, HORIZONTAL, Label

import smbus2

import time

import tm1637

# --- 1. Nastavení 7-segmentového displeje (TM1637) ---

# CLK na GPIO 21 (pin 40), DIO na GPIO 20 (pin 38)

displej_7seg = tm1637.TM1637(clk=21, dio=20)

displej_7seg.brightness(2) # Nastavení jasu (0 až 7)

displej_7seg.write([0, 0, 0, 0]) # Vymaže 7-segment při startu

# --- 2. Nastavení I2C LCD (Stejné jako minule) ---

I2C_ADRESA = 0x27 

bus = smbus2.SMBus(1)

def lcd_posli(bits, mode):

    bits_high = mode | (bits & 0xF0) | 0x08

    bits_low = mode | ((bits << 4) & 0xF0) | 0x08

    bus.write_byte(I2C_ADRESA, bits_high)

    lcd_toggle(bits_high)

    bus.write_byte(I2C_ADRESA, bits_low)

    lcd_toggle(bits_low)

def lcd_toggle(bits):

    time.sleep(0.0001)

    bus.write_byte(I2C_ADRESA, (bits | 0x04))

    time.sleep(0.0001)

    bus.write_byte(I2C_ADRESA, (bits & ~0x04))

    time.sleep(0.0001)

def lcd_init():

    lcd_posli(0x33, 0); lcd_posli(0x32, 0); lcd_posli(0x06, 0)

    lcd_posli(0x0C, 0); lcd_posli(0x28, 0); lcd_posli(0x01, 0)

    time.sleep(0.005)

def lcd_text(text, radek):

    text = text.ljust(16, " ")

    lcd_posli(radek, 0)

    for i in range(16):

        lcd_posli(ord(text[i]), 1)

# Start I2C displeje

try:

    lcd_init()

    lcd_text("Dashboard start!", 0x80)

except Exception as chyba:

    print(f"I2C Chyba LCD: {chyba}")

# --- 3. Funkce, co propojí PC s oběma displeji ---

def aktualizuj_vystup(hodnota):

    intenzita = int(hodnota)

    # A) Výstup na 16x2 LCD

    lcd_text("Hodnota z PC:", 0x80)

    lcd_text(f"--> {intenzita} %", 0xC0)

    # B) Výstup na 4-místný 7-segmentový displej

    displej_7seg.number(intenzita)

# --- 4. Grafické okno (Dashboard) ---

okno = Tk()

okno.title("Multi-Displej Dashboard")

okno.geometry("350x150")

nadpis = Label(okno, text="Posuvník (Vstup)", font=("Arial", 16))

nadpis.pack(pady=15)

posuvnik = Scale(okno, from_=0, to=100, orient=HORIZONTAL, command=aktualizuj_vystup, length=250)

posuvnik.pack(pady=10)

# Čisté vypnutí obou displejů při zavření křížkem

def pri_zavreni():

    try:

        lcd_posli(0x01, 0)

        displej_7seg.write([0, 0, 0, 0])

    except:

        pass

    okno.destroy()

okno.protocol("WM_DELETE_WINDOW", pri_zavreni)

okno.mainloop()
