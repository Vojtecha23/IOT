from tkinter import Tk, Scale, HORIZONTAL, Label
from gpiozero import PWMOutputDevice, Servo
# 1. Výstup: Bzučák na GPIO 18 (využije výchozí softwarové PWM)
bzucak = PWMOutputDevice(18)
# 2. Výstup: Servo na GPIO 13 (využije výchozí softwarové PWM)
servo = Servo(13)
def zmen_hlasitost(hodnota):
   # Bzučák bere intenzitu od 0.0 do 1.0
   intenzita = int(hodnota) / 100.0
   bzucak.value = intenzita
def zmen_uhel(hodnota):
   # Servo v gpiozero bere hodnoty od -1.0 do 1.0
   procenta = int(hodnota)
   uhel = (procenta / 50.0) - 1.0
   servo.value = uhel
# Nastavení grafického okna
okno = Tk()
okno.title("Duální Dashboard (Softwarové PWM)")
okno.geometry("350x250")
# --- Sekce pro Bzučák ---
Label(okno, text="Intenzita bzučáku", font=("Arial", 12, "bold")).pack(pady=(15, 0))
posuvnik_bzucak = Scale(okno, from_=0, to=100, orient=HORIZONTAL, command=zmen_hlasitost, length=250)
posuvnik_bzucak.pack(pady=(0, 10))
# --- Sekce pro Servo ---
Label(okno, text="Natočení serva", font=("Arial", 12, "bold")).pack(pady=(15, 0))
posuvnik_servo = Scale(okno, from_=0, to=100, orient=HORIZONTAL, command=zmen_uhel, length=250)
# Z bezpečnostních důvodů nastavíme servo rovnou doprostřed (hodnota 0.0)
posuvnik_servo.set(50)
posuvnik_servo.pack(pady=(0, 10))
# Spuštění aplikace
okno.mainloop()
